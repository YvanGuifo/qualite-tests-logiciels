\section{Introduction : la qualité logicielle}
% ============================================================================
\sectionframe{1. Qualité logicielle}{Pourquoi tester ? Que mesurer ? \quad\textbf{\textbar{}}\quad \textbf{40 min}}

% ---- 1.1 Définition --------------------------------------------------------
\begin{frame}[t]{Qu'est-ce que la qualité d'un logiciel ?}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.55\textwidth}
\vspace{0.1cm}
La \textbf{qualité logicielle} est l'aptitude d'un système à satisfaire :
\begin{itemize}\footnotesize
    \item ses besoins \textbf{fonctionnels} \\
    {\color{textsoft}\scriptsize (ce que fait le logiciel) ;}
    \item ses besoins \textbf{non-fonctionnels} \\
    {\color{textsoft}\scriptsize (fiabilité, sécurité, performance) ;}
    \item les \textbf{normes} et standards applicables.
\end{itemize}
\vspace{0.2cm}
\begin{beamercolorbox}[sep=0.5em,rounded=true]{block body}
\scriptsize\itshape
« Quality is never an accident; it is always the result of intelligent effort. »
\textcolor{textsoft}{--- John Ruskin}
\end{beamercolorbox}
\end{column}
\begin{column}{0.42\textwidth}
\vspace{0.1cm}
\textbf{Un logiciel peut être :}\\[0.2em]
\footnotesize
\oui{\textbf{Correct} fonctionnellement} \\[0.15em]
\non{\textbf{Lent} sous charge} \\[0.15em]
\non{\textbf{Difficile à maintenir}} \\[0.15em]
\non{\textbf{Vulnérable} aux injections SQL} \\[0.15em]
\oui{\textbf{Beau} à l'écran}\\[0.4em]
\textcolor{textsoft}{$\Rightarrow$ La qualité n'est pas binaire mais \textbf{multidimensionnelle}.}
\end{column}
\end{columns}
\takeaway{La qualité d'un logiciel ne se résume pas à « ça marche ». Il existe au moins 8 dimensions normalisées.}
\end{frame}

% ---- 1.2 Cas Knight Capital ------------------------------------------------
\begin{frame}[t]{Pourquoi se préoccuper de la qualité ? Cas réel}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.55\textwidth}
\vspace{0.1cm}
\textbf{\color{accent}Knight Capital --- 1\textsuperscript{er} août 2012}\\[0.3em]
\footnotesize
Société de trading américaine. Déploiement d'un nouveau logiciel sur 8 serveurs de production.\\[0.5em]
\non{\textbf{Erreur :}} sur \textbf{1 serveur sur 8}, un ancien code « test » (\texttt{Power Peg}) a été réactivé par erreur.\\[0.5em]
\non{\textbf{Conséquence :}} le système a passé \textbf{4 millions d'ordres} en 45 minutes sur 154 actions.\\[0.5em]
\non{\textbf{Perte :}} \textbf{440 millions de dollars}.\\[0.5em]
\non{\textbf{Fin :}} société rachetée le mois suivant.
\end{column}
\begin{column}{0.42\textwidth}
\vspace{0.1cm}
\textbf{Causes racines identifiées par la SEC :}\\[0.3em]
\footnotesize
\begin{itemize}\scriptsize
    \item Déploiement \textbf{non automatisé} (humain a oublié 1 serveur)
    \item Pas de \textbf{tests d'intégration} sur le déploiement complet
    \item Code mort réactivé sans détection
    \item Absence d'arrêt automatique sur anomalie
\end{itemize}
\vspace{0.3em}
\begin{beamercolorbox}[sep=0.4em,rounded=true]{block body}
\scriptsize\color{bad}\textbf{45 minutes pour perdre 440 M\$.}\\
{\color{textsoft}Source : SEC Release No.~70694 (2013).}
\end{beamercolorbox}
\end{column}
\end{columns}
\takeaway{Un défaut logiciel a un coût qui dépasse largement sa correction technique.}
\end{frame}

% ---- 1.7 Exercice ----------------------------------------------------------
\begin{frame}[t]{Exercice 1.1 --- Identifier les caractéristiques affectées}
\begin{exercice}
Une plateforme e-commerce affiche des \textbf{prix incorrects} lors de pics de charge\\
(> 1\,000 utilisateurs simultanés) mais fonctionne parfaitement avec 100 utilisateurs.\\
Le reste des fonctionnalités est impeccable.\\[0.4em]
\textbf{(a)} Quelle(s) caractéristique(s) ISO~25010 sont \textbf{affectées} ?\\
\textbf{(b)} Quelle(s) caractéristique(s) ne sont \textbf{pas} affectées ?\\
\textbf{(c)} Quelle stratégie de test permettrait de détecter ce problème ?
\end{exercice}
\vspace{0.4cm}
\centering\small\itshape\color{textsoft} Prenez 5~minutes en binôme.
\end{frame}

\begin{frame}[t]{Solution 1.1}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.50\textwidth}
\vspace{0.1cm}
\textbf{(a) Caractéristiques affectées :}
\begin{itemize}\footnotesize
    \item \textbf{Adéquation fonctionnelle} : prix incorrect = mauvaise exactitude.
    \item \textbf{Performance} : dégradation sous charge.
    \item \textbf{Fiabilité} : résultat non reproductible selon la charge.
\end{itemize}
\vspace{0.2em}\pause
\textbf{(b) Non affectées :}
\begin{itemize}\footnotesize
    \item Sécurité, Portabilité, Maintenabilité (sans information contraire).
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\vspace{0.1cm}\pause
\textbf{(c) Stratégie :}
\begin{itemize}\footnotesize
    \item \textbf{Tests de charge} : simuler 1\,000+ utilisateurs.
    \item \textbf{Tests de stress} : dépasser la limite.
    \item \textbf{Outils} : JMeter, k6, Locust.
    \item Tests d'intégration vérifiant la cohérence BDD/affichage sous charge.
\end{itemize}
\end{column}
\end{columns}
\takeaway{Un logiciel peut satisfaire toutes ses spécifications fonctionnelles et échouer sur des propriétés \textbf{émergentes} comme la performance.}
\end{frame}

\begin{tikzpicture}[node distance=1.0cm]
\node[chainbox, fill=bgaccent, draw=accent] (err) {\textbf{Erreur}\\\scriptsize action humaine};
\node[chainbox, fill=orangelight, draw=warn, right=of err] (def) {\textbf{Défaut}\\\scriptsize dans le code};
\node[chainbox, fill=redlight, draw=bad, right=of def] (pan) {\textbf{Panne}\\\scriptsize observée par l'utilisateur};
\draw[arrowstep] (err) -- (def);
\draw[arrowstep] (def) -- (pan);
\node[font=\tiny, text=textsoft, below=0.05cm of err]{(\textit{error / mistake})};
\node[font=\tiny, text=textsoft, below=0.05cm of def]{(\textit{defect / bug / fault})};
\node[font=\tiny, text=textsoft, below=0.05cm of pan]{(\textit{failure})};
\end{tikzpicture}
\vspace{0.6cm}
\footnotesize
\begin{itemize}\itemsep0.15em
\item Une \textbf{erreur} est faite par un humain (mauvaise interprétation, oubli, faute de frappe).
\item Elle peut produire un \textbf{défaut} dans le code, la documentation ou la configuration.
\item Le défaut, s'il est exécuté avec les bonnes données, provoque une \textbf{panne}.
\end{itemize}
\takeaway{Tous les défauts ne provoquent pas de panne : c'est l'exécution qui révèle le problème.}
\end{frame}

% ---- 2.2 Exemple concret ---------------------------------------------------
\begin{frame}[fragile,t]{Exemple concret : un défaut qui dort}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.55\textwidth}
\vspace{0.1cm}
\begin{lstlisting}[style=pythondark]
def diviser(a, b):
    # Le developpeur a oublie b == 0
    return a / b

# Cas nominal :
diviser(10, 2)   # -> 5.0  OK
diviser(7, 3.5)  # -> 2.0  OK

# Cas qui revele la panne :
diviser(5, 0)    # -> ZeroDivisionError
\end{lstlisting}
\end{column}
\begin{column}{0.42\textwidth}
\vspace{0.1cm}\footnotesize
\textbf{Décomposition :}\\[0.3em]
\textcolor{accent}{\textbf{Erreur}} : le développeur a oublié de penser au cas $b = 0$.\\[0.3em]
\textcolor{warn}{\textbf{Défaut}} : la fonction \texttt{diviser} ne vérifie pas $b$ avant la division.\\[0.3em]
\textcolor{bad}{\textbf{Panne}} : exception non gérée renvoyée à l'utilisateur quand $b = 0$.\\[0.3em]
\textcolor{textsoft}{Le défaut existait dès l'écriture, mais ne s'est révélé qu'à l'exécution avec une donnée particulière.}
\end{column}
\end{columns}
\takeaway{Tester, c'est créer des conditions d'exécution qui forcent les défauts dormants à devenir des pannes.}
\end{frame}

% ---- 2.3 V&V ---------------------------------------------------------------
\begin{frame}[t]{Vérification et validation : deux questions différentes}
\vspace{0.1cm}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\begin{beamercolorbox}[sep=0.6em,rounded=true]{block body}
\centering\textbf{\color{accent}Vérification}\\[0.2em]
\large\itshape « Avons-nous bien construit le logiciel ? »\\[0.4em]
\footnotesize\normalfont
On compare le produit à sa \textbf{spécification}.\\
$\Rightarrow$ \textbf{conformité} au cahier des charges.
\end{beamercolorbox}
\vspace{0.4em}\footnotesize
\textbf{Exemples :}
\begin{itemize}\scriptsize
\item Revue de code
\item Tests unitaires automatisés
\item Compilation sans warning
\item Vérification de types statique
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\begin{beamercolorbox}[sep=0.6em,rounded=true]{block body}
\centering\textbf{\color{accent2}Validation}\\[0.2em]
\large\itshape « Avons-nous construit le bon logiciel ? »\\[0.4em]
\footnotesize\normalfont
On compare le produit aux \textbf{besoins réels} de l'utilisateur.\\
$\Rightarrow$ \textbf{utilité} du produit.
\end{beamercolorbox}
\vspace{0.4em}\footnotesize
\textbf{Exemples :}
\begin{itemize}\scriptsize
\item Tests utilisateurs / UAT
\item Démo client
\item Mise en bêta sur un échantillon
\item Analyse de feedback en production
\end{itemize}
\end{column}
\end{columns}
\takeaway{Un logiciel \textbf{vérifié} peut être \textbf{invalide} : conforme à un cahier des charges erroné.}
\end{frame}

\begin{frame}[t]{Fil rouge : la librairie en ligne}
\footnotesize
\textbf{Scénario :} on développe un site de librairie. Spécification écrite : \textit{le panier additionne le prix des livres}.
\vspace{0.4cm}
\begin{tabular}{@{}c p{4.2cm} c p{4.4cm}@{}}
\toprule
\textbf{\#} & \textbf{Activité} & \textbf{Type} & \textbf{Justification} \\
\midrule
1 & Relecture du code de \texttt{Panier} par un collègue
& \textcolor{accent}{\bfseries Vérif.}
& On compare code et spec interne \\
2 & Démo aux libraires
& \textcolor{accent2}{\bfseries Valid.}
& On confronte au besoin réel \\
3 & Exécution \texttt{pytest} sur le module panier
& \textcolor{accent}{\bfseries Vérif.}
& Tests = spec encodée \\
4 & Test par 5 utilisateurs réels en boutique
& \textcolor{accent2}{\bfseries Valid.}
& Mesure de satisfaction client \\
5 & Conformité au cahier des charges
& \textcolor{accent}{\bfseries Vérif.}
& Document de référence existe \\
\bottomrule
\end{tabular}
\takeaway{Des tests unitaires qui passent garantissent la vérification --- pas la validation. Les deux sont nécessaires.}
\end{frame}

% ---- 2.5 Exercice 2 --------------------------------------------------------
\begin{frame}[fragile,t]{Exercice 2.1 --- Identifier erreur, défaut, panne}
\begin{exercice}
Un développeur écrit la fonction Python suivante :
\begin{lstlisting}[style=pythondark]
def moyenne(notes):
    # somme divisee par 10 (oubli de len(notes))
    return sum(notes) / 10
\end{lstlisting}
Un étudiant entre ses 12 notes (moyennes mensuelles).\\
Affichage : \textbf{« Moyenne = 14,5 / 20 »} alors qu'il devrait s'afficher \textbf{« 17,4 / 20 »}.\\[0.3em]
\textbf{Identifiez :} \textit{(a)} l'erreur, \textit{(b)} le défaut, \textit{(c)} la panne.
\end{exercice}
\end{frame}

\begin{frame}[t]{Solution 2.1}
\begin{itemize}\footnotesize
\item \textcolor{accent}{\textbf{Erreur} (humaine)} : le développeur a confondu « moyenne » et « somme divisée par un nombre fixe ». Il a peut-être codé en pensant à un cas particulier (10 notes) puis a oublié de généraliser.
\vspace{0.3em}
\item \textcolor{warn}{\textbf{Défaut} (dans le code)} : la ligne \texttt{return sum(notes)~/~10} ; le diviseur correct est \texttt{len(notes)}.
\vspace{0.3em}
\item \textcolor{bad}{\textbf{Panne} (observable)} : pour 12 notes dont la somme est 209, le programme affiche \textbf{20,9} (ou 14,5 dans l'énoncé) au lieu de \textbf{17,4}.
\end{itemize}
\vspace{0.4cm}
\textbf{Note pédagogique :} \footnotesize ce défaut n'est révélé que si on teste avec une liste \textit{de longueur différente de 10}. Une bonne suite de tests \textbf{paramétrée} l'aurait détecté immédiatement.
\takeaway{Un défaut peut produire 0 ou plusieurs pannes selon les données. Tester = explorer les entrées.}
\end{frame}

\section{La pyramide de tests et les principes FIRST}
% ============================================================================
\sectionframe{3. Pyramide de tests \& FIRST}{Organiser ses efforts \quad\textbf{\textbar{}}\quad \textbf{50 min}}

% ---- 3.1 Pourquoi structurer ----------------------------------------------
\begin{frame}[t]{Pourquoi structurer ses tests ?}
\footnotesize
On ne teste pas tout au même niveau. Imaginons une application web :\\[0.4em]
\begin{tabular}{@{}p{3.8cm} p{4.4cm} p{3.8cm}@{}}
\toprule
\textbf{\color{accent}Niveau} & \textbf{\color{accent}Ce qu'on teste} & \textbf{\color{accent}Vitesse} \\
\midrule
Unitaire & Une fonction, isolée & ms (très rapide) \\
Intégration & Plusieurs modules ensemble & 100 ms à quelques s \\
End-to-End (E2E) & Le système complet via UI & secondes à minutes \\
\bottomrule
\end{tabular}
\vspace{0.4cm}
\\
Si je n'écris que des tests E2E :
\begin{itemize}\scriptsize
\item ma suite met 30~min à tourner $\to$ feedback trop tardif ;
\item un test rouge ne dit pas où est le défaut ;
\item les tests sont fragiles (changement UI = tests cassés).
\end{itemize}
\takeaway{La \textbf{pyramide de tests} donne une heuristique d'investissement : beaucoup en bas, peu en haut.}
\end{frame}

% ---- 3.4 FIRST -------------------------------------------------------------
\begin{frame}[t,shrink=10]{Les principes FIRST d'un bon test unitaire}
\vspace{0.2cm}
\centering
\begin{tikzpicture}[node distance=0.45cm and 0.6cm, font=\footnotesize]
\node[firstletter] (F) {F};
\node[firstletter, right=of F] (I) {I};
\node[firstletter, right=of I] (R) {R};
\node[firstletter, right=of R] (S) {S};
\node[firstletter, right=of S] (T) {T};
\end{tikzpicture}
\vspace{0.5cm}
\footnotesize
\begin{tabular}{@{}p{1.4cm} p{2.6cm} p{8.4cm}@{}}
\toprule
\textbf{\color{accent}Lettre} & \textbf{\color{accent}Principe} & \textbf{\color{accent}Définition opérationnelle} \\
\midrule
\textbf{F} & \textbf{Fast} (rapide)
& Quelques ms par test. Toute la suite doit tourner en quelques secondes.\\
\textbf{I} & \textbf{Independent} (indépendant)
& Un test ne dépend pas d'un autre. Ordre d'exécution indifférent.\\
\textbf{R} & \textbf{Repeatable} (reproductible)
& Même résultat à chaque exécution, sur toute machine, sans dépendance externe.\\
\textbf{S} & \textbf{Self-Validating} (auto-valide)
& Le test affirme lui-même son verdict via \texttt{assert}. Pas d'inspection humaine.\\
\textbf{T} & \textbf{Timely} (au moment opportun)
& Écrit \textbf{avec} ou \textbf{avant} le code de production (anticipation $\to$ TDD).\\
\bottomrule
\end{tabular}
\vspace{0.2cm}
\\
\centering{\scriptsize\color{textsoft}Source : Beck (2002), Martin (2008).}
\takeaway{FIRST = mnémonique simple pour audit rapide d'une suite de tests existante.}
\end{frame}

% ---- 3.5 FIRST violations exemples -----------------------------------------
\begin{frame}[fragile,t]{FIRST : exemples de violations à reconnaître}
\footnotesize
\begin{tabular}{@{}p{1.2cm} p{10.4cm}@{}}
\toprule
\textbf{F} & Test qui fait une requête HTTP réelle vers \texttt{api.github.com} (lent, dépend du réseau).\\
\textbf{I} & Test \texttt{test\_b} suppose que \texttt{test\_a} a déjà créé un utilisateur dans la base.\\
\textbf{R} & Test qui utilise \texttt{datetime.now()} : passe le matin, échoue le soir.\\
\textbf{S} & Test qui affiche un résultat avec \texttt{print()} sans \texttt{assert} : passe toujours.\\
\textbf{T} & Tests écrits 6 mois après le code, sur du code intestable.\\
\bottomrule
\end{tabular}
\vspace{0.4cm}
\begin{lstlisting}[style=pythondark]
# Violation S : ce "test" passe TOUJOURS car aucun assert
def test_addition_pas_self_validating():
    resultat = 2 + 3
    print(resultat)  # Sortie visuelle, pas d'assertion
\end{lstlisting}
\takeaway{Un test sans \texttt{assert} est pire qu'un test absent : il donne une fausse confiance.}
\end{frame}

% ---- 3.6 Exercice 3 --------------------------------------------------------
\begin{frame}[t]{Exercice 3.1 --- Diagnostiquer une suite de tests}
\begin{exercice}
Une équipe vous montre sa suite de tests avec ces caractéristiques :\\[0.3em]
\begin{itemize}\footnotesize
\item 4 tests unitaires, 38 tests E2E via Selenium, 0 test d'intégration ;
\item la suite met 22~minutes à s'exécuter ;
\item parfois les tests passent, parfois ils échouent --- l'équipe relance ;
\item le test \texttt{test\_panier\_total} suppose que \texttt{test\_panier\_creation} a déjà été exécuté.
\end{itemize}
\vspace{0.3em}
\textbf{Question :} identifiez \textbf{tous les problèmes} en mobilisant pyramide + FIRST. Proposez 2 corrections prioritaires.
\end{exercice}
\end{frame}

\begin{frame}[t]{Solution 3.1}
\footnotesize
\begin{itemize}
\item \textbf{Pyramide inversée} (cornet de glace) : 4 unitaires vs 38 E2E. Coûteux, lent.
\item \textbf{Violation F} : 22 min de suite $\Rightarrow$ feedback impraticable en CI.
\item \textbf{Violation R} : tests « flaky » (passent / échouent au hasard) $\Rightarrow$ relance = perte de confiance.
\item \textbf{Violation I} : dépendance entre \texttt{test\_panier\_total} et \texttt{test\_panier\_creation}.
\end{itemize}
\vspace{0.4cm}
\textbf{Corrections prioritaires :}\\[0.2em]
\begin{enumerate}\footnotesize
\item \textcolor{accent}{Réinvestir} : remplacer 30 tests E2E par 100+ tests unitaires couvrant la même logique.
\item \textcolor{accent2}{Rendre chaque test autonome} : créer ses propres données via \textbf{fixtures}, pas via un autre test.
\end{enumerate}
\takeaway{Une suite saine = pyramide respectée + FIRST honoré sur chaque test pris isolément.}
\end{frame}

% ---- 4.1 Pourquoi un framework --------------------------------------------
\begin{frame}[fragile,t,shrink=10]{Pourquoi un framework de test ?}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.49\textwidth}
\textbf{\color{bad}Sans framework}
\begin{lstlisting}[style=pythondark]
def test_addition():

        print("ECHEC : 2+3 != 5")

\begin{lstlisting}[style=pythondark]
def test_addition():

\end{lstlisting}

$ pytest
collected 2 items
test_calc.py ..  [100%]
==== 2 passed in 0.02s ====
\end{lstlisting}
\scriptsize\color{textsoft} Découverte auto, bilan, code de retour pour la CI.
\end{column}
\end{columns}
\takeaway{Un framework apporte : découverte, isolation, reporting, intégration CI.}
\end{frame}

% ---- 4.3 Conventions -------------------------------------------------------
\begin{frame}[t]{Conventions de découverte}
\footnotesize
\texttt{pytest} suit des règles \textbf{simples et strictes} pour découvrir tests et assertions :\\[0.3em]
\begin{tabular}{@{}p{4.4cm} p{8cm}@{}}
\toprule
\textbf{\color{accent}Convention} & \textbf{\color{accent}Détail} \\
\midrule
Fichiers & Commencent par \texttt{test\_} ou finissent par \texttt{\_test.py} \\
Fonctions & Commencent par \texttt{test\_} \\
Classes (optionnel) & Commencent par \texttt{Test} et n'ont \emph{pas} de \texttt{\_\_init\_\_} \\
Assertions & Utilisent le mot-clé Python \texttt{assert} (pas de méthode spéciale) \\
Rapport d'échec & Affichage automatique avec valeurs intermédiaires \\
\bottomrule
\end{tabular}
\vspace{0.4cm}
\textbf{Pourquoi c'est élégant :} pas d'héritage de \texttt{TestCase} comme en \texttt{unittest}. On écrit du Python naturel.
\takeaway{Convention plutôt que configuration : on écrit des fonctions Python, le framework les trouve.}
\end{frame}

% ---- 4.4 Patron AAA --------------------------------------------------------
\begin{frame}[t]{Le patron AAA --- Arrange, Act, Assert}
\centering
\begin{tikzpicture}[node distance=1.1cm, font=\footnotesize]
\node[chainbox, fill=bgaccent, draw=accent, minimum width=2.8cm] (a) {\textbf{Arrange}\\\scriptsize préparer les données};
\node[chainbox, fill=bgaccent2, draw=accent2, minimum width=2.8cm, right=of a] (b) {\textbf{Act}\\\scriptsize exécuter le comportement};
\node[chainbox, fill=greenlight, draw=good, minimum width=2.8cm, right=of b] (c) {\textbf{Assert}\\\scriptsize vérifier le résultat};
\draw[arrowstep] (a) -- (b);
\draw[arrowstep] (b) -- (c);
\end{tikzpicture}
\\
\vspace{0.8cm}
\footnotesize
\begin{tabular}{@{}p{1.7cm} p{10.5cm}@{}}
\toprule
\textbf{Étape} & \textbf{Rôle} \\
\midrule
\textbf{Arrange} & On instancie les objets nécessaires, on prépare les entrées. \\
\textbf{Act} & On appelle \textbf{une seule fois} la méthode testée. \\
\textbf{Assert} & On vérifie via \texttt{assert} que le résultat est conforme à l'attendu. \\
\bottomrule
\end{tabular}
\takeaway{Un test = un comportement, structuré en 3 étapes visuellement séparées par des lignes vides.}
\end{frame}

% ---- 4.5 Fil rouge code ----------------------------------------------------
\begin{frame}[fragile,t,shrink=20]{Fil rouge : module \texttt{librairie.py}}
\begin{lstlisting}[style=pythondark]
# src/librairie.py
from dataclasses import dataclass

@dataclass
class Livre:
    isbn: str
    titre: str
    prix: float
    stock: int

class LibraireService:
    def __init__(self, catalogue: dict[str, Livre]):
        self.catalogue = catalogue

    def get_prix(self, isbn: str) -> float:
        return self.catalogue[isbn].prix

    def est_disponible(self, isbn: str, quantite: int = 1) -> bool:
        livre = self.catalogue.get(isbn)
        return livre is not None and livre.stock >= quantite
\end{lstlisting}
\takeaway{Ce module sera testé tout au long de la séance --- accumulez vos tests dans \texttt{tests/test\_librairie.py}.}
\end{frame}

\section{Cas limites, parametrize et tolérance flottants}
% ============================================================================
\sectionframe{5. Cas limites \& \texttt{parametrize}}{Tester intelligemment \quad\textbf{\textbar{}}\quad \textbf{45 min}}

% ---- 5.1 Pourquoi cas limites ---------------------------------------------
\begin{frame}[t]{Pourquoi les cas limites trouvent les bugs}
\footnotesize
La plupart des défauts se cachent aux \textbf{frontières} des domaines de valeurs.\\[0.4em]
\textbf{Heuristique des valeurs limites} : pour un paramètre dans $[\min, \max]$, testez :
\begin{itemize}\itemsep0.1em
\item $\min - 1$, $\min$, $\min + 1$
\item une valeur \og typique \fg{} au milieu
\item $\max - 1$, $\max$, $\max + 1$
\item les cas \textbf{spéciaux} : 0, négatif, chaîne vide, \texttt{None}, liste vide
\end{itemize}
\vspace{0.4cm}
\textbf{Cas réel :} \texttt{Ariane 5} --- conversion \texttt{double} (64 bits) en \texttt{short} (16 bits). \\
Valeur testée sur Ariane 4 : OK. Valeur sur Ariane 5 : dépassait $2^{15}$. \\
\textit{Aucun test n'avait exploré la limite supérieure dans le contexte d'Ariane 5.}
\takeaway{« Ça marche sur mes exemples » $\neq$ « ça marche sur les limites ». Tester les frontières est non négociable.}
\end{frame}

% ---- 5.2 Exemple valider_age ----------------------------------------------
\begin{frame}[fragile,t,shrink=10]{Exemple : la fonction \texttt{valider\_age}}
\begin{lstlisting}[style=pythondark]
# Specification : un client doit avoir entre 13 et 120 ans inclus
def valider_age(age: int) -> bool:
    return 13 <= age <= 120
\end{lstlisting}
\vspace{0.3cm}
\textbf{Quels cas tester ?}\\[0.3em]
\footnotesize
\begin{tabular}{@{}l l l l@{}}
\toprule
\textbf{Entrée} & \textbf{Catégorie} & \textbf{Attendu} & \textbf{Type de cas} \\
\midrule
12 & juste avant min & \texttt{False} & limite \\
13 & exactement min & \texttt{True}  & limite \\
14 & juste après min & \texttt{True}  & limite \\
50 & nominal & \texttt{True} & nominal \\
119 & juste avant max & \texttt{True} & limite \\
120 & exactement max & \texttt{True} & limite \\
121 & juste après max & \texttt{False} & limite \\
-5 & négatif & \texttt{False} & spécial \\
0  & zéro & \texttt{False} & spécial \\
\bottomrule
\end{tabular}
\takeaway{Une bonne suite explore systématiquement les 7+ cas, pas juste 50.}
\end{frame}

% ---- 5.3 Sans parametrize : duplication -----------------------------------
\begin{frame}[fragile,t]{Naïvement : 9 fonctions très similaires...}
\begin{lstlisting}[style=pythondark]
def test_valider_age_12_faux():  assert not valider_age(12)
def test_valider_age_13_vrai():  assert valider_age(13)
def test_valider_age_14_vrai():  assert valider_age(14)
def test_valider_age_50_vrai():  assert valider_age(50)
def test_valider_age_119_vrai(): assert valider_age(119)
def test_valider_age_120_vrai(): assert valider_age(120)
def test_valider_age_121_faux(): assert not valider_age(121)
def test_valider_age_neg5_faux(): assert not valider_age(-5)
def test_valider_age_0_faux():   assert not valider_age(0)
\end{lstlisting}
\vspace{0.2cm}
\textbf{Problèmes :}
\begin{itemize}\footnotesize
\item Duplication massive (9 fonctions ne diffèrent que par l'entrée).
\item Ajouter un cas = copier-coller un bloc.
\item Modifier le comportement attendu = modifier 9 endroits.
\end{itemize}
\takeaway{La duplication dans les tests trahit un manque de \textbf{paramétrage}.}
\end{frame}

% ---- 5.4 Parametrize ------------------------------------------------------
\begin{frame}[fragile,t,shrink=10]{La solution : \texttt{@pytest.mark.parametrize}}
\begin{lstlisting}[style=pythondark]
import pytest

@pytest.mark.parametrize("age, attendu", [
    (12,  False),  # juste avant min
    (13,  True),   # exactement min
    (14,  True),   # juste apres min
    (50,  True),   # nominal
    (119, True),   # juste avant max
    (120, True),   # exactement max
    (121, False),  # juste apres max
    (-5,  False),  # negatif
    (0,   False),  # zero
])
def test_valider_age(age, attendu):
    assert valider_age(age) == attendu
\end{lstlisting}
\vspace{0.2cm}
\footnotesize\texttt{pytest} génère \textbf{9 tests indépendants}, chacun avec son propre identifiant et son propre rapport.
\takeaway{Un seul corps de test, N jeux de données. Lisibilité maximale, duplication zéro.}
\end{frame}

% ---- 5.6 Le piège des flottants -------------------------------------------
\begin{frame}[fragile,t]{Piège classique : les flottants}
\begin{lstlisting}[style=pythondark]
>>> 0.1 + 0.2
0.30000000000000004
>>> 0.1 + 0.2 == 0.3
False
\end{lstlisting}
\vspace{0.3cm}
\textbf{Pourquoi ?} Représentation IEEE 754 binaire : \texttt{0.1} n'a pas de représentation exacte en base 2 (cf.\ IEEE Std 754-2019).\\[0.4em]
\footnotesize
\textbf{En production, conséquences :}
\begin{itemize}\scriptsize
\item Le total d'un panier de 0,99~\euro{} $\times$ 3 vaut 2,97 \textit{ou} 2,9700000000000004 selon les versions.
\item Les comparaisons strictes \texttt{==} sur des prix calculés sont fragiles.
\end{itemize}
\takeaway{N'utilisez jamais \texttt{==} sur des résultats flottants calculés. Utilisez une tolérance.}
\end{frame}

% ---- 5.7 pytest.approx -----------------------------------------------------
\begin{frame}[fragile,t,shrink=15]{Solution : \texttt{pytest.approx}}
\begin{lstlisting}[style=pythondark]
import pytest

def test_total_panier_avec_approx():
    service = LibraireService({
        "978-0": Livre("978-0", "Clean Code", 0.99, 10)
    })

    total = service.calculer_total("978-0", 3)

    # Tolerance relative par defaut : 1e-6
    assert total == pytest.approx(2.97)

    # Tolerance personnalisable :
    assert total == pytest.approx(2.97, rel=1e-3)
    assert total == pytest.approx(2.97, abs=0.01)
\end{lstlisting}
\vspace{0.3cm}
\footnotesize \texttt{pytest.approx} accepte des paramètres \texttt{rel} (relative) ou \texttt{abs} (absolue), ainsi que des listes/dictionnaires.
\takeaway{Pour tout calcul flottant : \texttt{== pytest.approx(valeur)}. Reflexe à intégrer dès aujourd'hui.}
\end{frame}

\begin{frame}[fragile,t]{Exercice 5.1 --- TVA paramétrée}

On vous demande de tester la fonction suivante :

    void setUp() {

\begin{frame}[t]{Fixtures : pytest vs JUnit 5}

\begin{tabular}{@{}p{4.8cm} p{4.0cm} p{4.0cm}@{}}
\toprule
\textbf{\color{accent}Objectif} & \textbf{\color{accent}pytest} & \textbf{\color{accent}JUnit 5} \\
\midrule
Préparer avant \textbf{chaque} test & \texttt{@pytest.fixture} +\\ argument injecté & \texttt{@BeforeEach} \\
\addlinespace[0.2em]

\addlinespace[0.2em]
Préparer une seule fois pour toute la classe & \texttt{scope="class"} & \texttt{@BeforeAll} (méthode \texttt{static}) \\
\addlinespace[0.2em]

\bottomrule
\end{tabular}

\footnotesize
Sources possibles : \texttt{@CsvSource}, \texttt{@ValueSource}, \texttt{@MethodSource}, \texttt{@CsvFileSource}.
\takeaway{JUnit 5 offre des \emph{ParameterizedTest} aussi expressifs que \texttt{pytest.mark.parametrize}.}
\end{frame}

% ---- 6.8 TP exercice ------------------------------------------------------
\begin{frame}[fragile,t]{Exercice TP --- Librairie en Java}
\begin{exercice}[Exercice 6.1 (sur machine, 20 min)]
On vous fournit la classe \texttt{LibraireService} en Java (équivalente à la version Python).\\[0.4em]
\textbf{Écrire en JUnit 5 :}\\[0.2em]
\textbf{(1)} Un test pour \texttt{getPrix("978-0")} retourne \texttt{29.90}.\\
\textbf{(2)} Un test pour \texttt{estDisponible("978-0", 3)} retourne \texttt{true} si stock = 5.\\
\textbf{(3)} Un test pour \texttt{estDisponible("978-0", 10)} retourne \texttt{false} si stock = 5.\\
\textbf{(4)} Un test paramétré \texttt{@CsvSource} couvrant la méthode \texttt{calculerTotal} avec : (1 livre, 1 unité), (3 unités), (0 unité).\\[0.3em]
\textbf{Contraintes :} utilisez \texttt{@BeforeEach}, respectez AAA et le nommage de méthodes.
\end{exercice}
\end{frame}

\begin{frame}[fragile,t,shrink=25]{Solution 6.1 (extrait)}
\begin{lstlisting}[style=javadark]
class LibraireServiceTest {
    private LibraireService service;

    @BeforeEach
    void setUp() {
        service = new LibraireService(Map.of(
            "978-0", new Livre("978-0", "Clean Code", 29.90, 5)
        ));
    }

    @Test
    void estDisponible_retourne_vrai_si_stock_suffisant() {
        assertTrue(service.estDisponible("978-0", 3));
    }

    @ParameterizedTest
    @CsvSource({"1, 29.90", "3, 89.70", "0, 0.0"})
    void calculerTotal_retourne_prix_multiplie(int qte, double attendu) {
        assertEquals(attendu, service.calculerTotal("978-0", qte), 0.001);
    }
}
\end{lstlisting}
\takeaway{Mêmes principes pédagogiques que pytest. Vos réflexes AAA et FIRST sont \textbf{transférables}.}
\end{frame}

\begin{frame}[t]{Synthèse de la séance 1}
\footnotesize
\textbf{6 concepts fondateurs maîtrisés :}\\[0.3em]
\begin{enumerate}\itemsep0.18em
\item \textbf{Qualité} = multidimensionnelle, normée par ISO/IEC 25010.
\item \textbf{Erreur / défaut / panne} = trois maillons distincts d'une chaîne.
\item \textbf{Vérification vs validation} = deux questions complémentaires.
\item \textbf{Pyramide de tests} = 70 / 20 / 10 entre unitaires, intégration, E2E.
\item \textbf{FIRST} = audit rapide de toute suite (Fast, Independent, Repeatable, Self-validating, Timely).
\item \textbf{Patron AAA} = squelette universel d'un test, en pytest comme en JUnit 5.
\end{enumerate}
\vspace{0.4cm}
\textbf{2 compétences techniques opérationnelles :}\\[0.2em]
\oui{\textbf{pytest}} : tests simples, \texttt{parametrize}, \texttt{approx}.\\
\oui{\textbf{JUnit 5}} : \texttt{@Test}, \texttt{@BeforeEach}, \texttt{@ParameterizedTest}, \texttt{assertThrows}.
\takeaway{Vous êtes équipé·e pour écrire et lire des tests dans deux écosystèmes industriels majeurs.}
\end{frame}
